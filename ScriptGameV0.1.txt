# 🔁 MAIN LOOP (CLEAN VERSION)

```python
score = 0
running = True

while running:
    screen.fill((135, 206, 235))

    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        if event.type == pygame.KEYDOWN:
            if event.key == pygame.K_SPACE:
                velocity = jump

    # Physics
    velocity += gravity
    y += velocity

    # Spawn coins
    if random.randint(1, 60) == 1:
        coins.append(spawn_coin())

    # Move coins
    for c in coins:
        c[0] -= 4

    # Draw louage
    screen.blit(louage, (x, y))

    # Draw coins
    for c in coins:
        screen.blit(dinar, (c[0], c[1]))

        if abs(c[0]-x) < 40 and abs(c[1]-y) < 40:
            score += 1
            coins.remove(c)

    # 🚓 POLICE CHASE (after score 5)
    if score >= 5:
        police_x += police_speed
        police_y = y

        screen.blit(police, (police_x, police_y))

        if abs(police_x - x) < 50:
            print("Caught by police!")
            running = False

    pygame.display.update()
    clock.tick(60)

pygame.quit()
```
